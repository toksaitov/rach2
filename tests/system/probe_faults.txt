# The faults tests/system/probe_faults.sh puts into the tree, one at a time,
# each under a row of the default suite that must fail with it. A block is
#
#   fault NAME     a name unique in this list
#   probe PROBE    the probe, as the runner names it
#   file PATH      the file the fault changes, from the top of the tree
#   expect WORDS   words the probe prints only when that row fails
#   old            lines of the file, which occur in it exactly once,
#   ...
#   new            and the lines that replace them
#   ...
#   end
#
# and a line starting with # outside a block is a comment. Most faults take
# a fix of milestone 54's review back, or bring back the defect its row is
# about; the comment before each group names the part of the system its
# faults are in.

# The times statx reports, which musl's stat calls on i386: whole, where the
# stat64 structure keeps the low 32 bits.

fault statx-time-signed32
probe statmusl
file rach2/sys/syscall_fs.c
expect STATMUSL FAIL (stat reports -2147483638 for 2147483658 seconds)
old
    sx->stx_mtime.tv_sec = st->st_mtime;
new
    sx->stx_mtime.tv_sec = (int32_t)st->st_mtime;
end

fault statx-time-unsigned32
probe statmusl
file rach2/sys/syscall_fs.c
expect STATMUSL FAIL (stat reports 4263431296 for -31536000 seconds)
old
    sx->stx_mtime.tv_sec = st->st_mtime;
new
    sx->stx_mtime.tv_sec = (uint32_t)st->st_mtime;
end

# utimensat_time64, which musl's utimensat calls on i386 for a time past
# 2^31 seconds, and the 64-bit seconds it reads.

fault utimensat-time64-missing
probe statmusl
file rach2/sys/arch/i686/systab.c
expect STATMUSL FAIL (utimensat to 2147483658 seconds failed)
old
    table[SYS_UTIMENSAT_TIME64] = sys_utimensat_time64;
new
    (void)sys_utimensat_time64;
end

fault utimensat-time64-seconds32
probe statmusl
file rach2/sys/syscall_fs.c
expect STATMUSL FAIL (stat reports -2147483638 for 2147483658 seconds)
old
        t[1].tv_sec = v[2];
new
        t[1].tv_sec = (long)v[2];
end
