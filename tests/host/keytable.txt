# keytable.txt: the bytes each key of a 105-key PC keyboard sends, as xterm
# sends them, which the kernel console and term both send.
#
# The bytes are xterm's, patch 412 (2026-10-11), from its input.c and
# ctlseqs.txt. X has xkeyboard-config's pc105 model with the us layout and no
# options, so Right Alt is Alt_R and the keypad's Shift inverts Num Lock, and
# without the levels that give the X server Ctrl+Alt with F1 to F12, Backspace
# and the keypad's operators, so those keys keep their keysyms. xterm runs in a
# UTF-8 locale with these resources and every other at its default:
#
#   XTerm*metaSendsEscape: true
#   XTerm*backarrowKey: false
#   XTerm*backarrowKeyIsErase: false
#   XTerm*omitTranslation: fullscreen,scroll-lock,select,shift-fonts
#   XTerm*openIm: false
#
# So Alt sends ESC before a key's character, Backspace sends DEL and
# Ctrl+Backspace BS, and of xterm's own uses of keys only Shift with Page Up
# and Page Down remain, which scroll the screen back. The defaults that matter
# are keyboardType unknown, deleteIsDEL maybe (Delete sends CSI 3 ~),
# modifyCursorKeys and modifyFunctionKeys 2, modifyKeypadKeys and
# modifyOtherKeys 0, and numLock true. The modifier parameter of a sequence is
# 1 plus 1 for Shift, 2 for Alt and 4 for Ctrl.
#
# A line is a mode line, a key line or a press line, and # starts a comment:
#
#   modes MODE[,MODE]...      the modes of the lines below; "modes -" for none
#   KEY C1 C2 C3 C4 C5 C6 C7 C8
#   press EVENT... = RESULT
#
# A key line gives what KEY sends under the modifiers of each column: none,
# Shift, Ctrl, Ctrl+Shift, Alt, Alt+Shift, Ctrl+Alt and Ctrl+Alt+Shift. Each
# modifier is held on the left, on the right and on both sides, so a cell is
# 1, 3, 9 or 27 cases: the modifiers are pressed in the order Shift, Ctrl and
# Alt, KEY is pressed and released, and the modifiers are released in the
# reverse order.
#
# A press line gives what a sequence of events sends in all: +KEY presses a
# key, -KEY releases it and KEY presses and releases it. The names shift, ctrl
# and alt stand for the left key, the right key and both, as in a key line.
#
# KEY is the key's name in <linux/input-event-codes.h>, lowercased without its
# KEY_ prefix: leftshift, kp7, 102nd. A cell or RESULT is one of
#
#   "..."     the bytes, with \e for ESC, \r, \n, \t, \b, \\, \" and \xHH
#   local     the terminal acts on the key and sends nothing
#   SIGINT, SIGQUIT, SIGTSTP
#             the signal the tty sends its foreground process group
#
# and a MODE one of
#
#   ckm       DECCKM set (CSI ? 1 h): the cursor keys in application mode
#   kpam      DECKPAM set (ESC =): the keypad in application mode
#   num       Num Lock on before the first event; off without it
#   caps      Caps Lock on before the first event; off without it
#   tty       the result is what a process reading the tty receives, with the
#             termios that cfmakeraw() leaves but for the flags named below,
#             Linux's default c_cc, and the terminal's bytes as above
#   icanon, icrnl, isig
#             that flag set, with tty
#
# tests/host/keytable.c reads the table; tests/host/Readme.md says who runs it.

# The line discipline: Enter and the signal keys through the tty, raw, with
# ICRNL, and cooked, each with ISIG on and off.
modes tty
press enter = "\r"
press kpenter = "\r"
press +ctrl m -ctrl = "\r"
press +ctrl j -ctrl = "\n"
press +alt enter -alt = "\e\r"
press +ctrl c -ctrl = "\x03"
press +ctrl backslash -ctrl = "\x1c"
press +ctrl z -ctrl = "\x1a"
press +ctrl d -ctrl = "\x04"
press backspace = "\x7f"

modes tty,isig
press enter = "\r"
press +ctrl c -ctrl = SIGINT
press +ctrl backslash -ctrl = SIGQUIT
press +ctrl z -ctrl = SIGTSTP
press +ctrl d -ctrl = "\x04"

modes tty,icrnl
press enter = "\n"
press kpenter = "\n"
press +ctrl m -ctrl = "\n"
press +ctrl j -ctrl = "\n"
press +alt enter -alt = "\e\n"

modes tty,icanon,icrnl,isig
press a b enter = "ab\n"
press enter = "\n"
press kpenter = "\n"
press a b backspace enter = "a\n"
press a +ctrl m -ctrl = "a\n"
press a +ctrl j -ctrl = "a\n"
press +ctrl c -ctrl = SIGINT
press +ctrl backslash -ctrl = SIGQUIT
press +ctrl z -ctrl = SIGTSTP

modes tty,icanon,icrnl
press +ctrl c -ctrl enter = "\x03\n"
press +ctrl backslash -ctrl enter = "\x1c\n"
press +ctrl z -ctrl enter = "\x1a\n"

modes tty,icanon
press a enter +ctrl j -ctrl = "a\r\n"
